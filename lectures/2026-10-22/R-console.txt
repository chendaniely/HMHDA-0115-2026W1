  pivot_longer(
    cols = starts_with("wk"),
    names_to = "week",
    names_prefix = "wk",

    values_to = "rank",

billboard_tidy |>
  group_by(week) |>
  summarize(average_rank = mean(rank)) |>

  geom_line()
